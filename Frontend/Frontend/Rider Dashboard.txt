<!DOCTYPE html>
<html>
<head>
  <title>Rider Dashboard</title>
  <link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/bootstrap/dist/css/bootstrap.min.css">
</head>
<body class="container mt-5">
  <h2>My Assigned Deliveries</h2>
  <div id="deliveries"></div>
  <button id="pickupBtn" class="btn btn-success mt-3">Mark as Picked Up</button>

  <script>
    async function loadDeliveries() {
      const res = await fetch('/api/riders/assigned?rider_id=1');
      const data = await res.json();
      document.getElementById('deliveries').innerText = JSON.stringify(data, null, 2);
    }

    document.getElementById('pickupBtn').addEventListener('click', async () => {
      const res = await fetch('/api/riders/update', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ rider_id: 1, status: 'Picked Up' })
      });
      const data = await res.json();
      alert("Status updated: " + data.status);
    });

    loadDeliveries();
  </script>
</body>
</html>
